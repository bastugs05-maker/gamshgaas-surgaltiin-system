async function login() { const email = document.getElementById("email").value.trim(); const password = document.getElementById("password").value; const btn = document.getElementById("loginBtn"); if (!email || !password) { showMessage("loginStatus", "И-мэйл болон нууц үгээ оруулна уу.", "error"); return; } btn.disabled = true; btn.textContent = "Нэвтэрч байна..."; showMessage("loginStatus", "Supabase холболтыг шалгаж байна..."); try { if (!window.supabase) { throw new Error("Supabase SDK ачаалагдсангүй."); } const result = await Promise.race([ supabase.auth.signInWithPassword({ email, password }), new Promise((_, reject) => setTimeout(() => reject(new Error("Холболтын хугацаа дууслаа. Интернэтээ шалгана уу.")), 15000) ) ]); if (result.error) throw result.error; currentUser = result.data.user; showMessage("loginStatus", "Амжилттай нэвтэрлээ!", "success"); await showApp(); } catch (err) { console.error("LOGIN ERROR:", err); showMessage("loginStatus", "Нэвтрэх алдаа: " + err.message, "error"); } finally { btn.disabled = false; btn.textContent = "Нэвтрэх"; } }
