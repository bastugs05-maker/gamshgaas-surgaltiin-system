async function login() { const email = document.getElementById("email").value.trim(); const password = document.getElementById("password").value; const status = document.getElementById("loginStatus"); const btn = document.querySelector("button[onclick='login()']"); if (!email || !password) { status.textContent = "И-мэйл болон нууц үгээ оруулна уу."; return; } status.textContent = "Нэвтэрч байна..."; if (btn) btn.disabled = true; try { const { data, error } = await supabase.auth.signInWithPassword({ email: email, password: password }); if (error) { status.textContent = "Нэвтрэх алдаа: " + error.message; return; } if (!data.user) { status.textContent = "Хэрэглэгчийн мэдээлэл олдсонгүй."; return; } currentUser = data.user; status.textContent = "Амжилттай нэвтэрлээ."; await showApp(); } catch (err) { status.textContent = "Системийн алдаа: " + (err.message || err); console.error("LOGIN ERROR:", err); } finally { if (btn) btn.disabled = false; } }
